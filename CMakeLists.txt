cmake_minimum_required(VERSION 3.25)
project(runyard VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(RUNYARD_BUILD_TESTS "Build automated tests" ON)
option(RUNYARD_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(RUNYARD_THREAD_SANITIZE "Enable thread sanitizer (separate build)" OFF)
if(RUNYARD_SANITIZE AND RUNYARD_THREAD_SANITIZE)
  message(FATAL_ERROR "Use separate address and thread sanitizer builds")
endif()

add_library(runyard_options INTERFACE)
target_compile_options(runyard_options INTERFACE -Wall -Wextra -Wpedantic)
if(RUNYARD_SANITIZE)
  target_compile_options(runyard_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(runyard_options INTERFACE -fsanitize=address,undefined)
elseif(RUNYARD_THREAD_SANITIZE)
  target_compile_options(runyard_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(runyard_options INTERFACE -fsanitize=thread)
endif()

find_package(Threads REQUIRED)
add_subdirectory(src)
if(RUNYARD_BUILD_TESTS)
  enable_testing()
  find_package(GTest CONFIG REQUIRED)
  include(GoogleTest)
  add_subdirectory(tests)
endif()
